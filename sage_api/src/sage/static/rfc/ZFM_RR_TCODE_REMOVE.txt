FUNCTION zfm_rr_tcode_remove.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(IV_ROLE_NAME) TYPE  AGR_NAME
*"     VALUE(IV_TCODE) TYPE  SYTCODE
*"     VALUE(IV_TEST_RUN) TYPE  CHAR1 DEFAULT SPACE
*"     VALUE(IV_REQUEST) TYPE  TRKORR OPTIONAL
*"  EXPORTING
*"     VALUE(EV_RETURN_TYPE) TYPE  BAPI_MTYPE
*"     VALUE(EV_RETURN_MSG) TYPE  BAPI_MSG
*"     VALUE(EV_TRANSPORT) TYPE  TRKORR
*"     VALUE(EV_REMOVED_COUNT) TYPE  INT4
*"     VALUE(ET_RETURN) TYPE  BAPIRET2_T
*"----------------------------------------------------------------------
  " Parameter names are fixed by executor.py (_execute_tcode_removal_via_custom_fm).
  " No classic exceptions: pyrfc would turn them into one error for the whole call.
  DATA(request) = VALUE lcl_rr_tcode_removal=>removal_request(
                    role      = to_upper( condense( iv_role_name ) )
                    tcode     = to_upper( condense( iv_tcode ) )
                    transport = iv_request ).
  DATA(removal) = NEW lcl_rr_tcode_removal( ).

  DATA(result) = COND lcl_rr_tcode_removal=>removal_result(
                   WHEN iv_test_run = abap_true
                   THEN removal->simulate( request )
                   ELSE removal->remove( request ) ).

  ev_return_type    = result-return_type.
  ev_return_msg     = result-message.
  ev_transport      = result-transport.
  ev_removed_count  = result-removed_values.
  et_return         = result-messages.
ENDFUNCTION.
